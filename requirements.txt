# Audio Flamingo 3 MIA release — Python dependencies.
#
# Install inside a conda environment (Python 3.10 recommended):
#
#   conda create -n af3 python=3.10 -y
#   conda activate af3
#   conda install -c nvidia cuda-toolkit -y
#   export CUDA_HOME=$CONDA_PREFIX
#   pip install -r requirements.txt
#   # flash-attn builds against the torch installed above, so install it after:
#   pip install flash-attn==2.7.3 --no-build-isolation
#   # triton 3.1.0 must also be installed after torch (see note below):
#   pip install triton==3.1.0
#
# Or just run ./environment_setup.sh, which performs all of the above.
#
# Notes:
# - Versions are pinned to the tested Audio Flamingo 3 environment.
# - torch==2.3.0 from PyPI is the CUDA 12.1 build on Linux x86_64.
# - The conda CUDA toolkit provides nvcc / CUDA_HOME, needed to build flash-attn
#   and for DeepSpeed to probe the GPU.
# - flash-attn is intentionally NOT listed here; see the comment above.
# - triton==3.1.0 is also installed separately, AFTER torch: torch==2.3.0
#   hard-pins triton==2.3.0, so co-resolving them fails. Installing triton last
#   only prints a benign "torch requires triton==2.3.0" warning. It is required
#   because llava/model/__init__.py imports the FP8 configs at module load, which
#   need triton.language.extra.cuda.libdevice (a triton 3.x path).

# --- PyTorch ---

# exists in triton 3.x. Listing triton==3.1.0 alongside torch==2.3.0 makes
# `pip install -r` fail with ResolutionImpossible, so (like flash-attn) it is
# installed as a separate post-install step; see the header and environment_setup.sh.
